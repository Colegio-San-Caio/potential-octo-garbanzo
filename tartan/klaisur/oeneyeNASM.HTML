[NASM BOX] - 16-bit - BLUH.XKTS grid 16,16 
; oeneyeNASM.asm - MOSFETQ font
; nasm -f bin oeneyeNASM.asm -o oeneyeOS.img
; qemu-system-i386 -fda oeneyeOS.img

[org 0x7c00]
[bits 16]
start:
  mov ax, 0x0003
  int 0x10
  mov si, msg
print:
  lodsb
  or al, al
  jz done
  mov ah, 0x0e
  mov bx, 0x0007
  int 0x10
  jmp print
done:
  hlt
msg db 'MOSFETQ - OENEYE - 16x16', 0
times 510-($-$$) db 0
dw 0xAA55

Build: nasm -f bin oeneyeNASM.asm -o oeneyeOS.img && qemu-system-i386 -fda oeneyeOS.img
